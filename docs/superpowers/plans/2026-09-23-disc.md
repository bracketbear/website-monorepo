# Disc Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A protoplanetary-disc toy on the Animation Lab's toys shelf: gas on a log-polar GPU grid, planets on the CPU that the gas feels and feeds, and a throw gesture that seeds them.

**Architecture:** One `PingPong` (RGBA32F, 512 φ × 192 ln r) holds Σ, v*r, v*φ and mass-eaten-this-frame. A single step shader runs three passes per substep selected by `uPass` (sources+sink, azimuthal transport with FARGO integer shift, radial transport). A CPU mirror in `physics.ts` implements the identical operator split on a small grid and is the only thing the tests run. Planets integrate on the CPU with leapfrog; a reduce pass hands back per-planet disc force, mass eaten, and the inner-edge flux.

**Tech Stack:** TypeScript, WebGL2 (GLSL ES 3.00), PIXI v8 for the blit and the orbit-preview line, vitest for the mirror.

**Spec:** `docs/superpowers/specs/2026-09-23-disc-design.md`

## Global Constraints

- Units: `GM★ = 1`, `R_IN = 0.25`, `R_OUT = 2.0`; `dt = 0.02`; wall clock `1.57` units/s; at most 4 substeps a frame.
- Grid: 512 (φ) × 192 (ln r) in production; texture x = φ index `j`, texture y = radial index `i`; array index `k = i * nphi + j`.
- State channels: `x = Σ`, `y = v_r`, `z = v_φ` (full, not deviation), `w = mass eaten this frame`.
- `h = H/r` is the only constant that sets pressure, softening (`0.6 h r_p`), and colour (`T = 3000 K · r^-½`).
- The mirror and the shader must implement the same stencils: clamped `i` at the radial edges, wrapped `j`; Courant numbers clamped to ±0.9; Σ floored at `1e-4 · Σ0` after transport.
- All controls are `debug: true`. Package convention: `manifest.ts`, `shaders.ts`, `physics.ts`, `physics.test.ts`, `discAnimation.ts`, `index.ts`.
- Vitest is configured at the repo root: run `npx vitest run packages/flateralus-pixi-animations/src/disc` (and `npx vitest run packages/flateralus-pixi/src/pointer-service`) from the repo root. The package manager is npm.
- Commit after every task; commit messages in the repo's voice (`feat(disc): ...`), no trailing attribution unless the session reminder asks for it.

```ts
/**
 * The disc, on the CPU. This is the twin of `shaders.ts`: the same
 * operator split on the same log-polar grid, run small so the tests can
 * measure what the GPU cannot be asked. Every stencil here has to match
 * the shader's exactly — clamped i, wrapped j, the same one-sided edges —
 * or the mirror stops being a mirror.
 *
 * Units: GM★ = 1 and r = 1 is the middle of the disc. The orbit there
 * takes 2π units.
 */

export const R_IN = 0.25;
export const R_OUT = 2.0;
/** Fixed simulation step, and the most substeps a frame may take. */
export const STEP = 0.02;
export const MAX_STEPS_PER_FRAME = 4;
/** Wall-clock rate: the orbit at r = 1 takes 4 s. */
export const UNITS_PER_SECOND = 1.57;
export const MAX_PLANETS = 8;
/** A thrown seed, about the Earth against the Sun. */
export const SEED_Q = 1e-5;
/** Past this a planet has left the scene. Inside R_IN it has fallen in. */
export const R_ESCAPE = 2.2;

export interface DiscParams {
  dt: number;
  /** H/r. Sets c_s, the softening, the wake's pitch and the colour. */
  h: number;
  /** Shakura–Sunyaev α: ν = α c_s H. */
  alpha: number;
  /** Disc mass as a fraction of the star's. */
  discMass: number;
  /** 1/τ_acc: the fraction of Σ inside a sink removed per unit time. */
  accRate: number;
  /** A sink is never smaller than this many cells, so a fresh seed eats. */
  sinkFloorCells: number;
}

export const DEFAULT_PARAMS: DiscParams = {
  dt: STEP,
  h: 0.05,
  alpha: 3e-3,
  discMass: 0.02,
  accRate: 0.03,
  sinkFloorCells: 1.5,
};

export interface Planet {
  x: number;
  y: number;
  vx: number;
  vy: number;
  /** Mass as a fraction of the star's. */
  q: number;
  age: number;
  /** Acceleration from the disc, held from the last reduce. */
  ax: number;
  ay: number;
}

export interface Disc {
  nr: number;
  nphi: number;
  dlnr: number;
  dphi: number;
  /** Cell-centre radii (nr), face radii (nr + 1), cell areas (nr). */
  r: Float32Array;
  rf: Float32Array;
  area: Float32Array;
  sig: Float32Array;
  vr: Float32Array;
  vp: Float32Array;
  eat: Float32Array;
  /** The other half of the ping-pong pair. */
  sig2: Float32Array;
  vr2: Float32Array;
  vp2: Float32Array;
  eat2: Float32Array;
  /** Σ at r = 1 before the tapers, and the density floor. */
  sigma0: number;
  floor: number;
  /** Mass that has left through the domain faces. */
  outflow: number;
}

/** Locally isothermal: c_s = h v_K, so c_s² = h²/r. */
export function cs2(r: number, h: number): number {
  return (h * h) / r;
}

function smooth01(t: number): number {
  const x = Math.max(0, Math.min(1, t));
  return x * x * (3 - 2 * x);
}

/** Σ ∝ 1/r, eased in over the first 0.06 and out over the last 10 %. */
export function sigmaProfile(r: number, sigma0: number): number {
  const tin = smooth01((r - R_IN) / 0.06);
  const tout = 1 - smooth01((r - 0.9 * R_OUT) / (0.1 * R_OUT));
  return (sigma0 / r) * tin * tout;
}

/**
 * Radial gradient of a ring quantity with the stencil the step uses:
 * central in the interior, one-sided at the two edges. `get(i)` must
 * clamp i itself where it is asked beyond the grid.
 */
export function gradR(
  get: (i: number) => number,
  r: Float32Array,
  i: number,
  nr: number
): number {
  if (i === 0) return (get(1) - get(0)) / (r[1] - r[0]);
  if (i === nr - 1) return (get(i) - get(i - 1)) / (r[i] - r[i - 1]);
  return (get(i + 1) - get(i - 1)) / (r[i + 1] - r[i - 1]);
}

export function createDisc(nr: number, nphi: number, p: DiscParams): Disc {
  const dlnr = Math.log(R_OUT / R_IN) / nr;
  const dphi = (2 * Math.PI) / nphi;
  const r = new Float32Array(nr);
  const rf = new Float32Array(nr + 1);
  const area = new Float32Array(nr);
  for (let i = 0; i <= nr; i++) rf[i] = R_IN * Math.exp(i * dlnr);
  for (let i = 0; i < nr; i++) {
    r[i] = R_IN * Math.exp((i + 0.5) * dlnr);
    area[i] = 0.5 * (rf[i + 1] * rf[i + 1] - rf[i] * rf[i]) * dphi;
  }
  const n = nr * nphi;
  const sigma0 = p.discMass / (2 * Math.PI * (R_OUT - R_IN));
  const floor = 1e-4 * sigma0;

  const prof = new Float64Array(nr);
  for (let i = 0; i < nr; i++)
    prof[i] = Math.max(floor, sigmaProfile(r[i], sigma0));
  // Equilibrium rotation from the discrete pressure gradient, so the disc
  // opens balanced against the operator it will be stepped with rather
  // than against the analytic profile.
  const P = (i: number) => {
    const ii = Math.max(0, Math.min(nr - 1, i));
    return cs2(r[ii], p.h) * prof[ii];
  };
  const vpr = new Float64Array(nr);
  for (let i = 0; i < nr; i++) {
    const dPdr = gradR(P, r, i, nr);
    const v2 = r[i] * (1 / (r[i] * r[i]) + dPdr / prof[i]);
    vpr[i] = Math.sqrt(Math.max(0, v2));
  }

  const D: Disc = {
    nr,
    nphi,
    dlnr,
    dphi,
    r,
    rf,
    area,
    sig: new Float32Array(n),
    vr: new Float32Array(n),
    vp: new Float32Array(n),
    eat: new Float32Array(n),
    sig2: new Float32Array(n),
    vr2: new Float32Array(n),
    vp2: new Float32Array(n),
    eat2: new Float32Array(n),
    sigma0,
    floor,
    outflow: 0,
  };
  for (let i = 0; i < nr; i++)
    for (let j = 0; j < nphi; j++) {
      const k = i * nphi + j;
      D.sig[k] = prof[i];
      D.vp[k] = vpr[i];
    }
  return D;
}

// ---- measures

/** The ring whose centre is nearest r. */
export function ringIndex(D: Disc, r: number): number {
  const f = Math.log(r / R_IN) / D.dlnr - 0.5;
  return Math.max(0, Math.min(D.nr - 1, Math.round(f)));
}

export function ringMean(D: Disc, i: number): number {
  let s = 0;
  for (let j = 0; j < D.nphi; j++) s += D.sig[i * D.nphi + j];
  return s / D.nphi;
}

export function totalMass(D: Disc): number {
  let m = 0;
  for (let i = 0; i < D.nr; i++) {
    let s = 0;
    for (let j = 0; j < D.nphi; j++) s += D.sig[i * D.nphi + j];
    m += s * D.area[i];
  }
  return m;
}

/** Relative m = 2 Fourier amplitude of Σ around ring i: the wake's signature. */
export function wakeM2(D: Disc, i: number): number {
  let c = 0;
  let s = 0;
  let m = 0;
  for (let j = 0; j < D.nphi; j++) {
    const phi = (j + 0.5) * D.dphi;
    const v = D.sig[i * D.nphi + j];
    c += v * Math.cos(2 * phi);
    s += v * Math.sin(2 * phi);
    m += v;
  }
  return Math.hypot(c, s) / Math.max(m, 1e-30);
}

/** Mean wake amplitude two scale heights inside and outside r_p. */
export function wakeAmplitude(D: Disc, rp: number, h: number): number {
  const H = h * rp;
  return (
    0.5 *
    (wakeM2(D, ringIndex(D, rp - 2 * H)) + wakeM2(D, ringIndex(D, rp + 2 * H)))
  );
}

/** Azimuth of the densest cell on ring i. */
export function peakPhi(D: Disc, i: number): number {
  let best = -Infinity;
  let at = 0;
  for (let j = 0; j < D.nphi; j++) {
    const v = D.sig[i * D.nphi + j];
    if (v > best) {
      best = v;
      at = j;
    }
  }
  return (at + 0.5) * D.dphi;
}

export function isHealthy(D: Disc): boolean {
  for (let k = 0; k < D.sig.length; k++) {
    if (!Number.isFinite(D.sig[k]) || D.sig[k] < 0) return false;
    if (!Number.isFinite(D.vr[k]) || !Number.isFinite(D.vp[k])) return false;
  }
  return true;
}
```

```ts
// ---- the operator split

export function swap(D: Disc): void {
  [D.sig, D.sig2] = [D.sig2, D.sig];
  [D.vr, D.vr2] = [D.vr2, D.vr];
  [D.vp, D.vp2] = [D.vp2, D.vp];
  [D.eat, D.eat2] = [D.eat2, D.eat];
}

export function hillRadius(pl: Planet): number {
  return Math.hypot(pl.x, pl.y) * Math.cbrt(pl.q / 3);
}

/** Plummer softening at 0.6 H, the convention the gap thresholds assume. */
export function softening(pl: Planet, h: number): number {
  return 0.6 * h * Math.hypot(pl.x, pl.y);
}

/**
 * Pressure gradient, star and planet gravity, centrifugal and Coriolis
 * terms, viscosity, and the accretion sink. Semi-implicit Euler: v_φ
 * first, then v_r with the new v_φ. Same stencils as `uPass == 0`.
 */
export function sourcePass(
  D: Disc,
  p: DiscParams,
  planets: readonly Planet[],
  clearEat: boolean
): void {
  const { nr, nphi, dlnr, dphi, r, sig, vr, vp, eat } = D;
  const h = p.h;
  const dt = p.dt;
  const accK = 1 - Math.exp(-p.accRate * dt);
  const at = (i: number, j: number) =>
    Math.max(0, Math.min(nr - 1, i)) * nphi + (((j % nphi) + nphi) % nphi);
  const np = planets.length;
  const px = new Float64Array(np);
  const py = new Float64Array(np);
  const pq = new Float64Array(np);
  const peps2 = new Float64Array(np);
  const prs = new Float64Array(np);
  for (let k = 0; k < np; k++) {
    const pl = planets[k];
    px[k] = pl.x;
    py[k] = pl.y;
    pq[k] = pl.q;
    const e = softening(pl, h);
    peps2[k] = e * e;
    prs[k] = 0.5 * hillRadius(pl);
  }

  for (let i = 0; i < nr; i++) {
    const ri = r[i];
    const rinv = 1 / ri;
    const c2 = cs2(ri, h);
    const c2p = cs2(r[Math.min(nr - 1, i + 1)], h);
    const c2m = cs2(r[Math.max(0, i - 1)], h);
    const nu = p.alpha * h * h * Math.sqrt(ri);
    for (let j = 0; j < nphi; j++) {
      const k = i * nphi + j;
      const s = Math.max(sig[k], D.floor);
      const P0 = c2 * sig[k];
      const Pp = c2p * sig[at(i + 1, j)];
      const Pm = c2m * sig[at(i - 1, j)];
      let dPdr: number;
      if (i === 0) dPdr = (Pp - P0) / (r[1] - r[0]);
      else if (i === nr - 1) dPdr = (P0 - Pm) / (r[i] - r[i - 1]);
      else dPdr = (Pp - Pm) / (r[i + 1] - r[i - 1]);
      const dPdphi =
        (c2 * (sig[at(i, j + 1)] - sig[at(i, j - 1)])) / (2 * dphi);

      const phi = (j + 0.5) * dphi;
      const cp = Math.cos(phi);
      const sp = Math.sin(phi);
      const x = ri * cp;
      const y = ri * sp;
      let fx = 0;
      let fy = 0;
      let s1 = sig[k];
      let removed = 0;
      for (let m = 0; m < np; m++) {
        const dx = x - px[m];
        const dy = y - py[m];
        const d2 = dx * dx + dy * dy;
        const d2e = d2 + peps2[m];
        const inv = pq[m] / (d2e * Math.sqrt(d2e));
        fx -= dx * inv;
        fy -= dy * inv;
        const rs = Math.max(prs[m], p.sinkFloorCells * ri * dphi);
        if (d2 < rs * rs) {
          const rem = s1 * accK;
          s1 -= rem;
          removed += rem;
        }
      }
      const arp = fx * cp + fy * sp;
      const app = -fx * sp + fy * cp;

      const vrC = vr[k];
      const vpC = vp[k];
      const lapVr =
        ((vr[at(i + 1, j)] - 2 * vrC + vr[at(i - 1, j)]) / (dlnr * dlnr) +
          (vr[at(i, j + 1)] - 2 * vrC + vr[at(i, j - 1)]) / (dphi * dphi)) *
        rinv *
        rinv;
      const lapVp =
        ((vp[at(i + 1, j)] - 2 * vpC + vp[at(i - 1, j)]) / (dlnr * dlnr) +
          (vp[at(i, j + 1)] - 2 * vpC + vp[at(i, j - 1)]) / (dphi * dphi)) *
        rinv *
        rinv;

      const vp1 =
        vpC + dt * (-vrC * vpC * rinv - dPdphi / (s * ri) + app + nu * lapVp);
      const vr1 =
        vrC +
        dt * (vp1 * vp1 * rinv - rinv * rinv - dPdr / s + arp + nu * lapVr);

      D.sig2[k] = s1;
      D.vr2[k] = vr1;
      D.vp2[k] = vp1;
      D.eat2[k] = clearEat ? removed : eat[k] + removed;
    }
  }
  swap(D);
}
```

```ts
/** Van Leer's harmonic-mean slope limiter: monotone, second order. */
export function vanLeer(am: number, a0: number, ap: number): number {
  const d1 = a0 - am;
  const d2 = ap - a0;
  const pr = d1 * d2;
  return pr > 0 ? (2 * pr) / (d1 + d2) : 0;
}

/** Upwind face value: the donor cell's value, extrapolated by its slope. */
export function faceValue(
  aL: number,
  sL: number,
  aR: number,
  sR: number,
  c: number
): number {
  return c > 0 ? aL + 0.5 * sL * (1 - c) : aR - 0.5 * sR * (1 + c);
}

const clampC = (c: number) => Math.max(-0.9, Math.min(0.9, c));

/**
 * Azimuthal transport with orbital advection (Masset 2000). Each ring is
 * shifted by the nearest whole number of cells to Ω dt / Δφ — an exact
 * permutation — and the remainder, plus the real deviation from Keplerian,
 * transports Σ, Σ v_r and Σ r v_φ in flux form. Same as `uPass == 1`.
 */
export function phiPass(D: Disc, p: DiscParams): void {
  const { nr, nphi, dphi, r, sig, vr, vp, eat } = D;
  const dt = p.dt;
  const q0 = new Float64Array(nphi);
  const q1 = new Float64Array(nphi);
  const q2 = new Float64Array(nphi);
  const u = new Float64Array(nphi);
  const F0 = new Float64Array(nphi);
  const F1 = new Float64Array(nphi);
  const F2 = new Float64Array(nphi);
  for (let i = 0; i < nr; i++) {
    const ri = r[i];
    const base = i * nphi;
    const om = Math.pow(ri, -1.5);
    const n = Math.round((om * dt) / dphi);
    const kres = (n * dphi * ri) / dt;
    const dx = ri * dphi;
    for (let j = 0; j < nphi; j++) {
      const k = base + j;
      q0[j] = sig[k];
      q1[j] = sig[k] * vr[k];
      q2[j] = sig[k] * ri * vp[k];
      u[j] = vp[k] - kres;
    }
    // Flux through the face between j and j+1.
    for (let j = 0; j < nphi; j++) {
      const jm = (j + nphi - 1) % nphi;
      const jp = (j + 1) % nphi;
      const jpp = (j + 2) % nphi;
      const c = clampC((0.5 * (u[j] + u[jp]) * dt) / dx);
      F0[j] =
        c *
        faceValue(
          q0[j],
          vanLeer(q0[jm], q0[j], q0[jp]),
          q0[jp],
          vanLeer(q0[j], q0[jp], q0[jpp]),
          c
        );
      F1[j] =
        c *
        faceValue(
          q1[j],
          vanLeer(q1[jm], q1[j], q1[jp]),
          q1[jp],
          vanLeer(q1[j], q1[jp], q1[jpp]),
          c
        );
      F2[j] =
        c *
        faceValue(
          q2[j],
          vanLeer(q2[jm], q2[j], q2[jp]),
          q2[jp],
          vanLeer(q2[j], q2[jp], q2[jpp]),
          c
        );
    }
    for (let j = 0; j < nphi; j++) {
      const jm = (j + nphi - 1) % nphi;
      const n0 = q0[j] - (F0[j] - F0[jm]);
      const n1 = q1[j] - (F1[j] - F1[jm]);
      const n2 = q2[j] - (F2[j] - F2[jm]);
      const dst = base + ((j + n) % nphi);
      const sg = Math.max(D.floor, n0);
      D.sig2[dst] = sg;
      D.vr2[dst] = n1 / sg;
      D.vp2[dst] = n2 / (sg * ri);
      D.eat2[dst] = eat[base + j];
    }
  }
  swap(D);
}

/**
 * Radial transport in flux form on the non-uniform cells, with mass fluxes
 * through faces of radius r_f so the area weighting is exact. Domain faces
 * are outflow: flux leaves with the upwind value, nothing enters. Same as
 * `uPass == 2`.
 */
export function rPass(D: Disc, p: DiscParams): void {
  const { nr, nphi, dphi, r, rf, area, sig, vr, vp, eat } = D;
  const dt = p.dt;
  const q0 = new Float64Array(nr);
  const q1 = new Float64Array(nr);
  const q2 = new Float64Array(nr);
  const F0 = new Float64Array(nr + 1);
  const F1 = new Float64Array(nr + 1);
  const F2 = new Float64Array(nr + 1);
  const dr = (i: number) => rf[i + 1] - rf[i];
  const cl = (i: number) => Math.max(0, Math.min(nr - 1, i));
  for (let j = 0; j < nphi; j++) {
    for (let i = 0; i < nr; i++) {
      const k = i * nphi + j;
      q0[i] = sig[k];
      q1[i] = sig[k] * vr[k];
      q2[i] = sig[k] * r[i] * vp[k];
    }
    const Q0 = (i: number) => q0[cl(i)];
    const Q1 = (i: number) => q1[cl(i)];
    const Q2 = (i: number) => q2[cl(i)];
    for (let f = 0; f <= nr; f++) {
      let w: number;
      let a0: number;
      let a1: number;
      let a2: number;
      if (f === 0) {
        w = Math.min(vr[j], 0);
        a0 = q0[0];
        a1 = q1[0];
        a2 = q2[0];
      } else if (f === nr) {
        w = Math.max(vr[(nr - 1) * nphi + j], 0);
        a0 = q0[nr - 1];
        a1 = q1[nr - 1];
        a2 = q2[nr - 1];
      } else {
        w = 0.5 * (vr[(f - 1) * nphi + j] + vr[f * nphi + j]);
        const c = clampC((w * dt) / (w > 0 ? dr(f - 1) : dr(f)));
        a0 = faceValue(
          Q0(f - 1),
          vanLeer(Q0(f - 2), Q0(f - 1), Q0(f)),
          Q0(f),
          vanLeer(Q0(f - 1), Q0(f), Q0(f + 1)),
          c
        );
        a1 = faceValue(
          Q1(f - 1),
          vanLeer(Q1(f - 2), Q1(f - 1), Q1(f)),
          Q1(f),
          vanLeer(Q1(f - 1), Q1(f), Q1(f + 1)),
          c
        );
        a2 = faceValue(
          Q2(f - 1),
          vanLeer(Q2(f - 2), Q2(f - 1), Q2(f)),
          Q2(f),
          vanLeer(Q2(f - 1), Q2(f), Q2(f + 1)),
          c
        );
      }
      const g = w * dt * rf[f] * dphi;
      F0[f] = g * a0;
      F1[f] = g * a1;
      F2[f] = g * a2;
    }
    D.outflow += -F0[0] + F0[nr];
    for (let i = 0; i < nr; i++) {
      const k = i * nphi + j;
      const n0 = q0[i] - (F0[i + 1] - F0[i]) / area[i];
      const n1 = q1[i] - (F1[i + 1] - F1[i]) / area[i];
      const n2 = q2[i] - (F2[i + 1] - F2[i]) / area[i];
      const sg = Math.max(D.floor, n0);
      D.sig2[k] = sg;
      D.vr2[k] = n1 / sg;
      D.vp2[k] = n2 / (sg * r[i]);
      D.eat2[k] = eat[k];
    }
  }
  swap(D);
}

/** One substep: sources and sink, azimuthal transport, radial transport. */
export function step(
  D: Disc,
  p: DiscParams,
  planets: readonly Planet[],
  clearEat: boolean
): void {
  sourcePass(D, p, planets, clearEat);
  phiPass(D, p);
  rPass(D, p);
}
```

```ts
// ---- the planets

/** Prograde circular velocity at (x, y). */
export function circularVelocity(x: number, y: number): [number, number] {
  const r = Math.hypot(x, y);
  const vK = 1 / Math.sqrt(r);
  return [(-y / r) * vK, (x / r) * vK];
}

function bodyAccel(
  planets: readonly Planet[],
  k: number,
  h: number
): [number, number] {
  const a = planets[k];
  const r = Math.hypot(a.x, a.y);
  const r3 = r * r * r;
  let ax = -a.x / r3 + a.ax;
  let ay = -a.y / r3 + a.ay;
  const e = softening(a, h);
  const e2 = e * e;
  for (let m = 0; m < planets.length; m++) {
    if (m === k) continue;
    const b = planets[m];
    const dx = b.x - a.x;
    const dy = b.y - a.y;
    const d2 = dx * dx + dy * dy + e2;
    const inv = b.q / (d2 * Math.sqrt(d2));
    ax += dx * inv;
    ay += dy * inv;
  }
  return [ax, ay];
}

/**
 * Kick-drift-kick leapfrog in the star's potential plus each other's, plus
 * the disc force held on each planet from the last reduce. Symplectic, so
 * the energy oscillates within (ω dt)²/8 and does not walk.
 */
export function stepBodies(planets: Planet[], dt: number, h: number): void {
  const n = planets.length;
  const acc: [number, number][] = [];
  for (let k = 0; k < n; k++) acc.push(bodyAccel(planets, k, h));
  for (let k = 0; k < n; k++) {
    const p = planets[k];
    p.vx += 0.5 * acc[k][0] * dt;
    p.vy += 0.5 * acc[k][1] * dt;
    p.x += p.vx * dt;
    p.y += p.vy * dt;
  }
  for (let k = 0; k < n; k++) {
    const [ax, ay] = bodyAccel(planets, k, h);
    const p = planets[k];
    p.vx += 0.5 * ax * dt;
    p.vy += 0.5 * ay * dt;
    p.age += dt;
  }
}

export function orbitEnergy(pl: Planet): number {
  return 0.5 * (pl.vx * pl.vx + pl.vy * pl.vy) - 1 / Math.hypot(pl.x, pl.y);
}

export interface Elements {
  bound: boolean;
  /** Semi-major axis; Infinity when unbound. */
  a: number;
  e: number;
  /** Argument of periapsis. */
  w: number;
  /** Semi-latus rectum, L². */
  p: number;
  L: number;
}

/** Orbital elements of a throw, from its position and velocity. */
export function elementsOf(
  x: number,
  y: number,
  vx: number,
  vy: number
): Elements {
  const r = Math.hypot(x, y);
  const E = 0.5 * (vx * vx + vy * vy) - 1 / r;
  const L = x * vy - y * vx;
  // Eccentricity vector: (v × L ẑ)/μ − r̂, with μ = 1.
  const ex = vy * L - x / r;
  const ey = -vx * L - y / r;
  const e = Math.hypot(ex, ey);
  return {
    bound: E < 0,
    a: E < 0 ? -1 / (2 * E) : Infinity,
    e,
    w: Math.atan2(ey, ex),
    p: L * L,
    L,
  };
}

/**
 * The conic r(θ) = p / (1 + e cos(θ − w)), sampled as xy pairs. A bound
 * orbit is the whole ellipse; an unbound one is the arc that stays inside
 * 1.5 R_ESCAPE, which is as far as the stage shows.
 */
export function orbitPath(el: Elements, n: number): Float32Array {
  const out: number[] = [];
  const limit = 1.5 * R_ESCAPE;
  let from = 0;
  let to = 2 * Math.PI;
  if (!el.bound) {
    const tInf = Math.acos(-1 / el.e);
    from = el.w - tInf;
    to = el.w + tInf;
  }
  for (let m = 0; m <= n; m++) {
    const th = from + ((to - from) * m) / n;
    const den = 1 + el.e * Math.cos(th - el.w);
    if (den <= 1e-6) continue;
    const r = el.p / den;
    if (r > limit) continue;
    out.push(r * Math.cos(th), r * Math.sin(th));
  }
  return Float32Array.from(out);
}

export type DiscEvent =
  | { type: 'merge'; q: number }
  | { type: 'fell'; q: number }
  | { type: 'escape'; q: number };

/**
 * Merge pairs inside the larger Hill radius, drop what has crossed the
 * cavity into the star, forget what has left. Mutates the array.
 */
export function settle(planets: Planet[], h: number): DiscEvent[] {
  void h;
  const events: DiscEvent[] = [];
  for (let a = 0; a < planets.length; a++) {
    for (let b = a + 1; b < planets.length; b++) {
      const A = planets[a];
      const B = planets[b];
      const d = Math.hypot(A.x - B.x, A.y - B.y);
      if (d < Math.max(hillRadius(A), hillRadius(B))) {
        const q = A.q + B.q;
        A.x = (A.x * A.q + B.x * B.q) / q;
        A.y = (A.y * A.q + B.y * B.q) / q;
        A.vx = (A.vx * A.q + B.vx * B.q) / q;
        A.vy = (A.vy * A.q + B.vy * B.q) / q;
        A.age = Math.max(A.age, B.age);
        A.q = q;
        planets.splice(b, 1);
        events.push({ type: 'merge', q });
        b--;
      }
    }
  }
  for (let k = planets.length - 1; k >= 0; k--) {
    const r = Math.hypot(planets[k].x, planets[k].y);
    if (r < R_IN) {
      events.push({ type: 'fell', q: planets[k].q });
      planets.splice(k, 1);
    } else if (r > R_ESCAPE) {
      events.push({ type: 'escape', q: planets[k].q });
      planets.splice(k, 1);
    }
  }
  return events;
}
```

```ts
// ---- what the disc hands back: the CPU twins of the reduce pass

/** Acceleration of a planet from the gas, softened as the gas is. */
export function discForce(D: Disc, pl: Planet, h: number): [number, number] {
  const e = softening(pl, h);
  const e2 = e * e;
  let ax = 0;
  let ay = 0;
  for (let i = 0; i < D.nr; i++) {
    const A = D.area[i];
    const ri = D.r[i];
    for (let j = 0; j < D.nphi; j++) {
      const phi = (j + 0.5) * D.dphi;
      const dx = ri * Math.cos(phi) - pl.x;
      const dy = ri * Math.sin(phi) - pl.y;
      const d2 = dx * dx + dy * dy + e2;
      const inv = (D.sig[i * D.nphi + j] * A) / (d2 * Math.sqrt(d2));
      ax += dx * inv;
      ay += dy * inv;
    }
  }
  return [ax, ay];
}

/** The eat channel, summed within the sink plus two cells of slop. */
export function eatenBy(D: Disc, pl: Planet, p: DiscParams): number {
  let m = 0;
  for (let i = 0; i < D.nr; i++) {
    const ri = D.r[i];
    const rs =
      Math.max(0.5 * hillRadius(pl), p.sinkFloorCells * ri * D.dphi) +
      2 * ri * D.dphi;
    for (let j = 0; j < D.nphi; j++) {
      const phi = (j + 0.5) * D.dphi;
      const dx = ri * Math.cos(phi) - pl.x;
      const dy = ri * Math.sin(phi) - pl.y;
      if (dx * dx + dy * dy < rs * rs) m += D.eat[i * D.nphi + j] * D.area[i];
    }
  }
  return m;
}

/** Mass per unit time through the cavity edge; negative is inward. */
export function innerFlux(D: Disc): number {
  let f = 0;
  for (let j = 0; j < D.nphi; j++)
    f += D.sig[j] * Math.min(D.vr[j], 0) * D.rf[0] * D.dphi;
  return f;
}

/** Component of an acceleration along the planet's prograde direction. */
export function tangentialForce(pl: Planet, ax: number, ay: number): number {
  const r = Math.hypot(pl.x, pl.y);
  return (-pl.y * ax + pl.x * ay) / r;
}
```

```ts
import { MAX_PLANETS } from './physics';

/** Slots in the reduce target: one per planet, then the star's row. */
export const REDUCE_W = MAX_PLANETS + 1;

/**
 * Shared prelude: the log-polar grid, clamped-and-wrapped fetch, the van
 * Leer pieces. `S(i, j)` is `at(i, j)` in physics.ts; `rAt`, `rfAt`,
 * `areaAt` are `D.r`, `D.rf`, `D.area`.
 */
const GRID = `
    uniform sampler2D uState;
    uniform float uRIn, uDlnr, uDphi;
    ivec2 G;
    vec4 S(int i, int j) {
      j = ((j % G.x) + G.x) % G.x;
      i = clamp(i, 0, G.y - 1);
      return texelFetch(uState, ivec2(j, i), 0);
    }
    float rAt(int i) { return uRIn * exp((float(i) + 0.5) * uDlnr); }
    float rfAt(int i) { return uRIn * exp(float(i) * uDlnr); }
    float areaAt(int i) { float a = rfAt(i), b = rfAt(i + 1); return 0.5 * (b * b - a * a) * uDphi; }
    float drAt(int i) { return rfAt(i + 1) - rfAt(i); }
    float vanLeer(float am, float a0, float ap) {
      float d1 = a0 - am, d2 = ap - a0, pr = d1 * d2;
      return pr > 0.0 ? 2.0 * pr / (d1 + d2) : 0.0;
    }
    float faceValue(float aL, float sL, float aR, float sR, float c) {
      return c > 0.0 ? aL + 0.5 * sL * (1.0 - c) : aR - 0.5 * sR * (1.0 + c);
    }
    float clampC(float c) { return clamp(c, -0.9, 0.9); }
`;

export const STEP_FS = `#version 300 es
    precision highp float;
    precision highp int;
    in vec2 vUV;
    out vec4 fragColor;
    ${GRID}
    uniform int uPass;
    uniform float uDt, uH, uAlpha, uAccK, uSinkFloor, uFloor;
    uniform int uClearEat;
    uniform int uPlanetN;
    uniform vec4 uPlanet[${MAX_PLANETS}];
    float cs2At(int i) { return uH * uH / rAt(clamp(i, 0, G.y - 1)); }

    // sourcePass()
    vec4 sources(int i, int j) {
      vec4 c = S(i, j);
      float r = rAt(i), rinv = 1.0 / r;
      float s = max(c.x, uFloor);
      float P0 = cs2At(i) * c.x;
      float Pp = cs2At(i + 1) * S(i + 1, j).x;
      float Pm = cs2At(i - 1) * S(i - 1, j).x;
      float dPdr;
      if (i == 0) dPdr = (Pp - P0) / (rAt(1) - rAt(0));
      else if (i == G.y - 1) dPdr = (P0 - Pm) / (rAt(i) - rAt(i - 1));
      else dPdr = (Pp - Pm) / (rAt(i + 1) - rAt(i - 1));
      float dPdphi = cs2At(i) * (S(i, j + 1).x - S(i, j - 1).x) / (2.0 * uDphi);

      float phi = (float(j) + 0.5) * uDphi;
      float cp = cos(phi), sp = sin(phi);
      vec2 x = vec2(r * cp, r * sp);
      vec2 f = vec2(0.0);
      float s1 = c.x, removed = 0.0;
      for (int m = 0; m < ${MAX_PLANETS}; m++) {
        if (m >= uPlanetN) break;
        vec4 pl = uPlanet[m];
        float eps = 0.6 * uH * pl.w;
        vec2 d = x - pl.xy;
        float d2 = dot(d, d);
        float d2e = d2 + eps * eps;
        f -= pl.z * d / (d2e * sqrt(d2e));
        float rh = pl.w * pow(pl.z / 3.0, 1.0 / 3.0);
        float rs = max(0.5 * rh, uSinkFloor * r * uDphi);
        if (d2 < rs * rs) { float rem = s1 * uAccK; s1 -= rem; removed += rem; }
      }
      float arp = f.x * cp + f.y * sp;
      float app = -f.x * sp + f.y * cp;

      float nu = uAlpha * uH * uH * sqrt(r);
      vec4 ip = S(i + 1, j), im = S(i - 1, j), jp = S(i, j + 1), jm = S(i, j - 1);
      float lapVr = ((ip.y - 2.0 * c.y + im.y) / (uDlnr * uDlnr) + (jp.y - 2.0 * c.y + jm.y) / (uDphi * uDphi)) * rinv * rinv;
      float lapVp = ((ip.z - 2.0 * c.z + im.z) / (uDlnr * uDlnr) + (jp.z - 2.0 * c.z + jm.z) / (uDphi * uDphi)) * rinv * rinv;
      float vp1 = c.z + uDt * (-c.y * c.z * rinv - dPdphi / (s * r) + app + nu * lapVp);
      float vr1 = c.y + uDt * (vp1 * vp1 * rinv - rinv * rinv - dPdr / s + arp + nu * lapVr);
      float eat = uClearEat == 1 ? removed : c.w + removed;
      return vec4(s1, vr1, vp1, eat);
    }

    // phiPass(): the output cell j took its value from js = j - n.
    vec4 phiPass(int i, int j) {
      float r = rAt(i);
      float shift = pow(r, -1.5) * uDt / uDphi;
      int n = int(floor(shift + 0.5));
      float kres = float(n) * uDphi * r / uDt;
      float dx = r * uDphi;
      int js = j - n;
      vec4 s[5];
      float q0[5], q1[5], q2[5], u[5];
      for (int m = 0; m < 5; m++) {
        s[m] = S(i, js - 2 + m);
        q0[m] = s[m].x;
        q1[m] = s[m].x * s[m].y;
        q2[m] = s[m].x * r * s[m].z;
        u[m] = s[m].z - kres;
      }
      float cL = clampC(0.5 * (u[1] + u[2]) * uDt / dx);
      float cR = clampC(0.5 * (u[2] + u[3]) * uDt / dx);
      float n0 = q0[2] - (cR * faceValue(q0[2], vanLeer(q0[1], q0[2], q0[3]), q0[3], vanLeer(q0[2], q0[3], q0[4]), cR)
                        - cL * faceValue(q0[1], vanLeer(q0[0], q0[1], q0[2]), q0[2], vanLeer(q0[1], q0[2], q0[3]), cL));
      float n1 = q1[2] - (cR * faceValue(q1[2], vanLeer(q1[1], q1[2], q1[3]), q1[3], vanLeer(q1[2], q1[3], q1[4]), cR)
                        - cL * faceValue(q1[1], vanLeer(q1[0], q1[1], q1[2]), q1[2], vanLeer(q1[1], q1[2], q1[3]), cL));
      float n2 = q2[2] - (cR * faceValue(q2[2], vanLeer(q2[1], q2[2], q2[3]), q2[3], vanLeer(q2[2], q2[3], q2[4]), cR)
                        - cL * faceValue(q2[1], vanLeer(q2[0], q2[1], q2[2]), q2[2], vanLeer(q2[1], q2[2], q2[3]), cL));
      float sg = max(n0, uFloor);
      return vec4(sg, n1 / sg, n2 / (sg * r), s[2].w);
    }

    // rPass(): faces i (below) and i + 1 (above); domain faces are outflow.
    vec4 rPass(int i, int j) {
      vec4 s[5];
      float q0[5], q1[5], q2[5];
      for (int m = 0; m < 5; m++) {
        int ii = clamp(i - 2 + m, 0, G.y - 1);
        s[m] = S(ii, j);
        q0[m] = s[m].x;
        q1[m] = s[m].x * s[m].y;
        q2[m] = s[m].x * rAt(ii) * s[m].z;
      }
      float wL, aL0, aL1, aL2;
      if (i == 0) {
        wL = min(s[2].y, 0.0); aL0 = q0[2]; aL1 = q1[2]; aL2 = q2[2];
      } else {
        wL = 0.5 * (s[1].y + s[2].y);
        float c = clampC(wL * uDt / (wL > 0.0 ? drAt(i - 1) : drAt(i)));
        aL0 = faceValue(q0[1], vanLeer(q0[0], q0[1], q0[2]), q0[2], vanLeer(q0[1], q0[2], q0[3]), c);
        aL1 = faceValue(q1[1], vanLeer(q1[0], q1[1], q1[2]), q1[2], vanLeer(q1[1], q1[2], q1[3]), c);
        aL2 = faceValue(q2[1], vanLeer(q2[0], q2[1], q2[2]), q2[2], vanLeer(q2[1], q2[2], q2[3]), c);
      }
      float wR, aR0, aR1, aR2;
      if (i == G.y - 1) {
        wR = max(s[2].y, 0.0); aR0 = q0[2]; aR1 = q1[2]; aR2 = q2[2];
      } else {
        wR = 0.5 * (s[2].y + s[3].y);
        float c = clampC(wR * uDt / (wR > 0.0 ? drAt(i) : drAt(i + 1)));
        aR0 = faceValue(q0[2], vanLeer(q0[1], q0[2], q0[3]), q0[3], vanLeer(q0[2], q0[3], q0[4]), c);
        aR1 = faceValue(q1[2], vanLeer(q1[1], q1[2], q1[3]), q1[3], vanLeer(q1[2], q1[3], q1[4]), c);
        aR2 = faceValue(q2[2], vanLeer(q2[1], q2[2], q2[3]), q2[3], vanLeer(q2[2], q2[3], q2[4]), c);
      }
      float gL = wL * uDt * rfAt(i) * uDphi;
      float gR = wR * uDt * rfAt(i + 1) * uDphi;
      float A = areaAt(i);
      float n0 = q0[2] - (gR * aR0 - gL * aL0) / A;
      float n1 = q1[2] - (gR * aR1 - gL * aL1) / A;
      float n2 = q2[2] - (gR * aR2 - gL * aL2) / A;
      float sg = max(n0, uFloor);
      return vec4(sg, n1 / sg, n2 / (sg * rAt(i)), s[2].w);
    }

    void main() {
      G = textureSize(uState, 0);
      ivec2 ij = ivec2(gl_FragCoord.xy);
      int j = ij.x, i = ij.y;
      fragColor = uPass == 0 ? sources(i, j) : (uPass == 1 ? phiPass(i, j) : rPass(i, j));
    }`;

/**

 * The last column: (innerFlux on row 0, mass, 0, 0). The CPU sums rows.

export const REDUCE_FS = `#version 300 es
    precision highp float;
    precision highp int;
    in vec2 vUV;
    out vec4 fragColor;
    ${GRID}
    uniform float uH, uSinkFloor;
    uniform int uPlanetN;
    uniform vec4 uPlanet[${MAX_PLANETS}];
    void main() {
      G = textureSize(uState, 0);
      int slot = int(gl_FragCoord.x);
      int i = int(gl_FragCoord.y);
      float r = rAt(i), A = areaAt(i);
      vec4 acc = vec4(0.0);
      if (slot < uPlanetN) {
        vec4 pl = uPlanet[slot];
        float eps = 0.6 * uH * pl.w;
        float eps2 = eps * eps;
        float rh = pl.w * pow(pl.z / 3.0, 1.0 / 3.0);
        float rs = max(0.5 * rh, uSinkFloor * r * uDphi) + 2.0 * r * uDphi;
        for (int j = 0; j < G.x; j++) {

          vec2 d = vec2(r * cos(phi), r * sin(phi)) - pl.xy;
          float d2 = dot(d, d);
          float d2e = d2 + eps2;
          acc.xy += (c.x * A) * d / (d2e * sqrt(d2e));
          if (d2 < rs * rs) acc.z += c.w * A;
        }
      } else if (slot == ${MAX_PLANETS}) {

          if (i == 0) acc.x += c.x * min(c.y, 0.0) * rfAt(0) * uDphi;
        }
      }
      fragColor = acc;
    }`;

/**
 * Everything on screen is a projection of Σ and v. Colour is the disc's
 * temperature T = 3000 K · r^-½ through a Planck fit, brightened where the
 * gas is being compressed (the wake is a weak shock); brightness is the
 * optical depth 1 − exp(−κΣ). The star's light is its accretion flux plus
 * a flare when a planet falls in. Planets are dots.
 */
export const DISPLAY_FS = `#version 300 es
    precision highp float;
    precision highp int;
    in vec2 vUV;
    out vec4 fragColor;
    ${GRID}
    uniform vec2 uRes;
    uniform float uTime, uGrain, uROut, uKappa, uView, uStarL;
    uniform int uPlanetN;
    uniform vec4 uPlanet[${MAX_PLANETS}];
    uniform vec3 uInk, uOrange;
    const float TAU = 6.28318530718;
    const float T0 = 3000.0;

    float hash21(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123); }
    float luma(vec3 c) { return dot(c, vec3(0.2126, 0.7152, 0.0722)); }

    // Bilinear in (ln r, φ), wrapping in φ by hand: the seam must not show.
    vec4 sampleDisc(float r, float phi) {
      float fi = log(r / uRIn) / uDlnr - 0.5;
      float fj = phi / uDphi - 0.5;
      int i0 = int(floor(fi)), j0 = int(floor(fj));
      float ti = fi - float(i0), tj = fj - float(j0);
      vec4 a = mix(S(i0, j0), S(i0, j0 + 1), tj);
      vec4 b = mix(S(i0 + 1, j0), S(i0 + 1, j0 + 1), tj);
      return mix(a, b, ti);
    }

    // Blackbody chromaticity, Helland's fit, valid 1000–40000 K.
    vec3 kelvin(float T) {
      float t = clamp(T, 1000.0, 40000.0) / 100.0;
      vec3 c;
      c.r = t <= 66.0 ? 1.0 : clamp(1.2929 * pow(t - 60.0, -0.1332), 0.0, 1.0);
      c.g = t <= 66.0 ? clamp(0.3901 * log(t) - 0.6318, 0.0, 1.0)
                      : clamp(1.1298 * pow(t - 60.0, -0.0755), 0.0, 1.0);
      c.b = t >= 66.0 ? 1.0 : (t <= 19.0 ? 0.0 : clamp(0.5432 * log(t - 10.0) - 1.1963, 0.0, 1.0));
      return c;
    }

    void main() {
      G = textureSize(uState, 0);
      vec3 ground = luma(uInk) < luma(uOrange) ? uInk : uOrange;
      vec2 p = (vUV - 0.5) * vec2(uRes.x / uRes.y, 1.0) * 2.0 * uView;
      float r = length(p);
      float phi = atan(p.y, p.x);
      if (phi < 0.0) phi += TAU;

      vec3 col = ground;
      if (r > uRIn && r < uROut) {
        vec4 c = sampleDisc(r, phi);
        float rp = r * exp(uDlnr), rm = r * exp(-uDlnr);
        vec4 cp = sampleDisc(min(rp, uROut * 0.999), phi);
        vec4 cm = sampleDisc(max(rm, uRIn * 1.001), phi);
        vec4 jp = sampleDisc(r, mod(phi + uDphi, TAU));
        vec4 jm = sampleDisc(r, mod(phi - uDphi + TAU, TAU));
        float div = (rp * cp.y - rm * cm.y) / ((rp - rm) * r) + (jp.z - jm.z) / (2.0 * uDphi * r);
        float om = pow(r, -1.5);
        float T = T0 * inversesqrt(r) * (1.0 + 0.5 * max(0.0, -div / om));
        float tau = 1.0 - exp(-uKappa * c.x);
        float L = pow(T / T0, 4.0);
        L = L / (1.0 + L);
        vec3 emit = kelvin(T) * L;
        // The outer taper fades into the ground so there is no rim.
        float edge = smoothstep(uROut, uROut * 0.92, r);
        col = mix(ground, emit, tau * edge);
      }

      // The star: a core and a glare whose strength is the accretion flux.
      float core = smoothstep(0.05, 0.02, r);
      float glare = uStarL * 0.003 / (r * r + 0.003);
      col += vec3(1.0, 0.93, 0.8) * (core + glare);

      for (int m = 0; m < ${MAX_PLANETS}; m++) {
        if (m >= uPlanetN) break;
        vec4 pl = uPlanet[m];
        float d = length(p - pl.xy);
        float dot_ = smoothstep(pl.z, pl.z * 0.5, d);
        float halo = 0.25 * pl.z * pl.z / (d * d + pl.z * pl.z);
        col += vec3(0.85, 0.9, 1.0) * (dot_ + halo);
      }

      col += (hash21(floor(gl_FragCoord.xy * 0.9) + vec2(fract(uTime * 7.0) * 13.0)) - 0.5) * 0.05 * uGrain;
      vec2 v = vUV * 2.0 - 1.0;
      col *= 0.92 + 0.08 * smoothstep(1.7, 0.3, dot(v, v));
      fragColor = vec4(col, 1.0);
    }`;
```

```ts
import * as PIXI from 'pixi.js';
import { PAL } from '@bracketbear/flateralus';
import { PixiAnimation } from '@bracketbear/flateralus-pixi';
import { PingPong } from '../shared/ping-pong';
import { setColor } from '../shared/shader-harness';
import { prefersReducedMotion } from '../shared/reduced-motion';
import { MANIFEST, type DiscControlValues } from './manifest';
import {
  STEP_FS,
  REDUCE_FS,
  DISPLAY_FS,
  STEP_UNIFORMS,
  REDUCE_UNIFORMS,
  DISP_UNIFORMS,
  REDUCE_W,
} from './shaders';
import {
  DEFAULT_PARAMS,
  MAX_PLANETS,
  MAX_STEPS_PER_FRAME,
  R_IN,
  R_OUT,
  SEED_Q,
  STEP,
  UNITS_PER_SECOND,
  circularVelocity,
  createDisc,
  elementsOf,
  orbitPath,
  settle,
  stepBodies,
  type DiscParams,
  type Planet,
} from './physics';

/** Production grid: 512 in φ, 192 in ln r. Cells are the same shape. */
const NPHI = 512;
const NR = 192;
/**
 * Steps run before the first frame, and how many per frame while that
 * happens. Ten orbits at r = 1 so the seeded planet's wake has wrapped
 * and settled. Real drivers only, no stamped spiral. The sink is off
 * during the pre-roll so nothing vanishes before the planet can be paid.
 */
const PREROLL_STEPS = 3140;
const PREROLL_PER_FRAME = 200;
/** Disc radius that fills the shorter screen dimension. */
const R_VIEW = 2.15;
/** Drag scale: one view radius of drag is 1.5 v_K at r = 1. */
const THROW_K = 1.5 / R_VIEW;
/** Under this many pixels a press-and-release is a click: circular orbit. */
const CLICK_PX = 6;
/** Seconds for a fall-in flare to decay. */
const FLARE_TAU = 1.5;
const PREVIEW_POINTS = 240;

interface Hold {
  /** Seed position in disc units. */
  ox: number;
  oy: number;
  /** Where the press landed, in pixels, for the click threshold. */
  px: number;
  py: number;
}

/**
 * A protoplanetary disc: gas integrated on the GPU on a log-polar grid,
 * planets on the CPU that the gas feels through the step shader and that
 * feel the gas through one reduce a frame. The pointer is not in the
 * system; only what it throws is.
 */
export class DiscAnimation extends PixiAnimation<typeof MANIFEST> {
  private pp: PingPong | null = null;
  private failed = false;
  private sprite: PIXI.Sprite | null = null;
  private texture: PIXI.Texture | null = null;
  private preview: PIXI.Graphics | null = null;

  private time = 0;
  private acc = 0;
  private preroll = 0;
  private seedKey = '';
  private reduced = false;
  private sigma0 = 0;
  private floor = 0;
  private planets: Planet[] = [];
  private hold: Hold | null = null;
  private flare = 0;
  private starFlux = 0;
  private planetBuf = new Float32Array(MAX_PLANETS * 4);

  constructor(initialControls?: Partial<DiscControlValues>) {
    super(MANIFEST, initialControls);
  }

  onInit(): void {
    this.time = 0;
    this.acc = 0;
    this.preroll = 0;
    this.seedKey = '';
    this.planets = [];
    this.hold = null;
    this.flare = 0;
    this.starFlux = 0;
  }

  private paramsOf(controls: DiscControlValues): DiscParams {
    return {
      ...DEFAULT_PARAMS,
      h: Number(controls.aspect),
      alpha: Number(controls.alpha),
      discMass: Number(controls.discMass),
      accRate: Number(controls.accretionRate),
    };
  }

  /** Build the opening disc from the mirror and hand it to the GPU. */
  private seed(pp: PingPong, p: DiscParams): void {
    const D = createDisc(NR, NPHI, p);
    const data = new Float32Array(NR * NPHI * 4);
    for (let k = 0; k < NR * NPHI; k++) {
      data[k * 4] = D.sig[k];
      data[k * 4 + 1] = D.vr[k];
      data[k * 4 + 2] = D.vp[k];
    }
    if (pp.sw === NPHI && pp.sh === NR) pp.reseed(data);
    else pp.initState(NPHI, NR, data);
    this.sigma0 = D.sigma0;
    this.floor = D.floor;
    // One planet already in it, at half the gap-opening mass: a strong
    // wake with something left to finish.
    const [vx, vy] = circularVelocity(1.1, 0);
    this.planets = [{ x: 1.1, y: 0, vx, vy, q: 2e-4, age: 0, ax: 0, ay: 0 }];
    this.hold = null;
    this.flare = 0;
    this.starFlux = 0;
    this.acc = 0;
    this.preroll = PREROLL_STEPS;
    this.reduced = prefersReducedMotion();
  }

  private packPlanets(forDisplay: boolean): number {
    const b = this.planetBuf;
    b.fill(0);
    const n = Math.min(MAX_PLANETS, this.planets.length);
    for (let k = 0; k < n; k++) {
      const pl = this.planets[k];
      b[k * 4] = pl.x;
      b[k * 4 + 1] = pl.y;
      if (forDisplay) {
        b[k * 4 + 2] = 0.02 * Math.cbrt(pl.q / 1e-4);
      } else {
        b[k * 4 + 2] = pl.q;
        b[k * 4 + 3] = Math.hypot(pl.x, pl.y);
      }
    }
    return n;
  }

  private setGrid(
    gl: WebGL2RenderingContext,
    L: Record<string, WebGLUniformLocation | null>
  ): void {
    gl.uniform1f(L.uRIn!, R_IN);
    gl.uniform1f(L.uDlnr!, Math.log(R_OUT / R_IN) / NR);
    gl.uniform1f(L.uDphi!, (2 * Math.PI) / NPHI);
  }

  /** One substep: three passes on the gas, then the planets. */
  private doStep(
    pp: PingPong,
    p: DiscParams,
    clearEat: boolean,
    sinkOn: boolean
  ): void {
    const n = this.packPlanets(false);
    const accK = sinkOn ? 1 - Math.exp(-p.accRate * p.dt) : 0;
    for (let pass = 0; pass < 3; pass++) {
      pp.step((gl, L) => {
        this.setGrid(gl, L);
        gl.uniform1i(L.uPass!, pass);
        gl.uniform1f(L.uDt!, p.dt);
        gl.uniform1f(L.uH!, p.h);
        gl.uniform1f(L.uAlpha!, p.alpha);
        gl.uniform1f(L.uAccK!, accK);
        gl.uniform1f(L.uSinkFloor!, p.sinkFloorCells);
        gl.uniform1f(L.uFloor!, this.floor);
        gl.uniform1i(L.uClearEat!, clearEat ? 1 : 0);
        gl.uniform1i(L.uPlanetN!, n);
        gl.uniform4fv(L.uPlanet!, this.planetBuf);
      });
    }
    stepBodies(this.planets, p.dt, p.h);
    for (const ev of settle(this.planets, p.h)) {
      if (ev.type === 'fell') this.flare += 4 * Math.cbrt(ev.q / 1e-4);
    }
  }

  /** The disc's answer for this frame: force on each planet, what it ate, the star's flux. */
  private readBack(pp: PingPong, p: DiscParams): void {
    const n = this.packPlanets(false);
    const rows = pp.reduce(REDUCE_W, NR, (gl, L) => {
      this.setGrid(gl, L);
      gl.uniform1f(L.uH!, p.h);
      gl.uniform1f(L.uSinkFloor!, p.sinkFloorCells);
      gl.uniform1i(L.uPlanetN!, n);
      gl.uniform4fv(L.uPlanet!, this.planetBuf);
    });
    if (!rows) return;
    for (let k = 0; k < n; k++) {
      let ax = 0;
      let ay = 0;
      let eaten = 0;
      for (let i = 0; i < NR; i++) {
        const o = (i * REDUCE_W + k) * 4;
        ax += rows[o];
        ay += rows[o + 1];
        eaten += rows[o + 2];
      }
      const pl = this.planets[k];
      if (Number.isFinite(ax) && Number.isFinite(ay)) {
        pl.ax = ax;
        pl.ay = ay;
      }
      if (Number.isFinite(eaten) && eaten > 0) pl.q += eaten;
    }
    const flux = rows[(0 * REDUCE_W + MAX_PLANETS) * 4];
    if (Number.isFinite(flux)) this.starFlux = -flux;
  }

  private drawPreview(sw: number, sh: number, vx: number, vy: number): void {
    const g = this.preview!;
    g.clear();
    if (!this.hold) return;
    const scale = Math.min(sw, sh) / (2 * R_VIEW);
    const el = elementsOf(this.hold.ox, this.hold.oy, vx, vy);
    const path = orbitPath(el, PREVIEW_POINTS);
    if (path.length < 4) return;
    const toX = (x: number) => sw / 2 + x * scale;
    const toY = (y: number) => sh / 2 - y * scale;
    // Solid where it flies, dashed where it would cross the cavity.
    for (let m = 2; m < path.length; m += 2) {
      const r = Math.hypot(path[m], path[m + 1]);
      const seg = m / 2;
      if (r < R_IN && seg % 2 === 0) continue;
      g.moveTo(toX(path[m - 2]), toY(path[m - 1]));
      g.lineTo(toX(path[m]), toY(path[m + 1]));
    }
    g.stroke({ width: 1, color: 0xffffff, alpha: el.bound ? 0.55 : 0.35 });
    g.circle(toX(this.hold.ox), toY(this.hold.oy), 3);
    g.fill({ color: 0xffffff, alpha: 0.8 });
  }

  onUpdate(
    app: PIXI.Application,
    controls: DiscControlValues,
    deltaTime: number
  ): void {
    if (this.failed) return;
    if (!this.pp) {
      const pp = PingPong.create({
        simFrag: STEP_FS,
        dispFrag: DISPLAY_FS,
        // Σ is walked by 1e-3 of itself a step; half floats stripe it.
        fullPrecision: true,
        simUniforms: STEP_UNIFORMS,
        dispUniforms: DISP_UNIFORMS,
        reduceFrag: REDUCE_FS,
        reduceUniforms: REDUCE_UNIFORMS,
      });
      if (!pp) {
        this.failed = true;
        return;
      }
      this.pp = pp;
    }
    const pp = this.pp;
    const p = this.paramsOf(controls);
    const dt = Math.min(0.05, deltaTime);
    this.time += dt;
    const sw = app.screen.width;
    const sh = app.screen.height;

    const seedKey = `${controls.discMass}|${controls.aspect}|${controls.reseed}`;
    if (seedKey !== this.seedKey) {
      this.seed(pp, p);
      this.seedKey = seedKey;
    }

    if (!this.sprite) {
      this.texture = PIXI.Texture.from(pp.canvas);
      this.sprite = new PIXI.Sprite(this.texture);
      this.getRoot().addChild(this.sprite);
      this.preview = new PIXI.Graphics();
      this.getRoot().addChild(this.preview);
    }
    const dpr = Math.min(2, window.devicePixelRatio || 1);
    const W = Math.max(2, Math.round(sw * dpr));
    const H = Math.max(2, Math.round(sh * dpr));
    if (pp.canvas.width !== W || pp.canvas.height !== H) {
      pp.canvas.width = W;
      pp.canvas.height = H;
      this.texture?.source.resize(W, H);
    }
    this.sprite.width = sw;
    this.sprite.height = sh;

    // ---- the throw
    const scale = Math.min(sw, sh) / (2 * R_VIEW);
    const toDisc = (x: number, y: number): [number, number] => [
      (x - sw / 2) / scale,
      (sh / 2 - y) / scale,
    ];
    const pointer = this.pointer;
    let pvx = 0;
    let pvy = 0;
    if (pointer) {
      while (pointer.clicks.length) {
        const c = pointer.clicks.shift()!;
        if (this.planets.length >= MAX_PLANETS) continue;
        const [ox, oy] = toDisc(c.x, c.y);
        this.hold = { ox, oy, px: c.x, py: c.y };
      }
      if (this.hold && (!pointer.mouse.down || !pointer.mouse.active)) {
        // Released elsewhere, or left the stage: the queue says which.
        if (!pointer.releases.length) this.hold = null;
      }
      if (this.hold && pointer.mouse.active) {
        const throwScale = this.reduced ? 0.35 : 1;
        pvx = ((pointer.mouse.x - this.hold.px) / scale) * THROW_K * throwScale;
        pvy =
          (-(pointer.mouse.y - this.hold.py) / scale) * THROW_K * throwScale;
      }
      while (pointer.releases.length) {
        const rel = pointer.releases.shift()!;
        if (!this.hold) continue;
        const drag = Math.hypot(rel.x - this.hold.px, rel.y - this.hold.py);
        let vx: number;
        let vy: number;
        if (drag < CLICK_PX) {
          [vx, vy] = circularVelocity(this.hold.ox, this.hold.oy);
        } else {
          const throwScale = this.reduced ? 0.35 : 1;
          vx = ((rel.x - this.hold.px) / scale) * THROW_K * throwScale;
          vy = (-(rel.y - this.hold.py) / scale) * THROW_K * throwScale;
        }
        this.planets.push({
          x: this.hold.ox,
          y: this.hold.oy,
          vx,
          vy,
          q: SEED_Q,
          age: 0,
          ax: 0,
          ay: 0,
        });
        this.hold = null;
      }
    }
    if (this.hold && drag2(this.hold, pointer) < CLICK_PX) {
      [pvx, pvy] = circularVelocity(this.hold.ox, this.hold.oy);
    }
    this.drawPreview(sw, sh, pvx, pvy);

    // ---- the pre-roll, outside the accumulator so the cap cannot eat it
    let pre = PREROLL_PER_FRAME;
    while (this.preroll > 0 && pre-- > 0) {
      this.preroll--;
      this.doStep(pp, p, true, false);
    }

    // ---- fixed-step integration, capped so a stall cannot spiral
    const rate =
      UNITS_PER_SECOND * Number(controls.speed) * (this.reduced ? 0.25 : 1);
    this.acc = Math.min(this.acc + dt * rate, MAX_STEPS_PER_FRAME * STEP);
    let first = true;
    let stepped = false;
    while (this.acc >= STEP) {
      this.acc -= STEP;
      this.doStep(pp, p, first, true);
      first = false;
      stepped = true;
    }
    if (stepped) this.readBack(pp, p);
    this.flare *= Math.exp(-dt / FLARE_TAU);

    // ---- display
    const n = this.packPlanets(true);
    // Flux reference: Σ at r = 1 moving inward at the sound speed around
    // the cavity's circumference. Real inflow is a fraction of it.
    const fluxRef = this.sigma0 * p.h * 2 * Math.PI * R_IN;
    const starL =
      1 + this.flare + Math.max(0, Math.min(2, this.starFlux / fluxRef));
    pp.display(W, H, (gl, L) => {
      this.setGrid(gl, L);
      gl.uniform2f(L.uRes!, W, H);
      gl.uniform1f(L.uTime!, this.time);
      gl.uniform1f(L.uGrain!, Number(controls.grain));
      gl.uniform1f(L.uROut!, R_OUT);
      gl.uniform1f(L.uKappa!, 1.5 / this.sigma0);
      gl.uniform1f(L.uView!, R_VIEW);
      gl.uniform1f(L.uStarL!, starL);
      gl.uniform1i(L.uPlanetN!, n);
      gl.uniform4fv(L.uPlanet!, this.planetBuf);
      setColor(gl, L.uInk!, PAL.ink);
      setColor(gl, L.uOrange!, PAL.orange);
    });
    this.texture?.source.update();
  }

  onDestroy(): void {
    this.pp?.destroy();
    this.pp = null;
    this.failed = false;
    this.sprite = null;
    this.preview = null;
    this.texture?.destroy(true);
    this.texture = null;
    super.onDestroy();
  }
}

/** Pixels the pointer has moved since the press, or 0 with no pointer. */
function drag2(
  hold: Hold,
  pointer: { mouse: { x: number; y: number } } | null
): number {
  if (!pointer) return 0;
  return Math.hypot(pointer.mouse.x - hold.px, pointer.mouse.y - hold.py);
}
```
